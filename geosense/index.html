const fetch = require('node-fetch');

const url = 'http://localhost:8000/api/v1/trigger-emergency';

  event_source: 'Node.js Agent',
  severity: 'CRITICAL',
  message: 'System critical signal broadcasted.',

};

fetch(url, {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'X-API-KEY': 'Geosense_DevPower_Secure_Secret_2026'
  },
  body: JSON.stringify(payload)
})
.then(res => res.json())
.then(json => console.log(json))
.catch(err => console.error(err));
